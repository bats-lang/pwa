      - name: Run the app on an emulator (attempt 1 of 2)
        id: emulator1
        continue-on-error: true
        uses: reactivecircus/android-emulator-runner@PIN
        env: &emulator-env
          SMOKE_TEXT: ${{ inputs.smoke-test-text }}
          SMOKE_SHARE_FILE: ${{ inputs.smoke-share-file }}
          SMOKE_SHARE_TYPE: ${{ inputs.smoke-share-type }}
          SMOKE_SHARE_TEXT: ${{ inputs.smoke-share-text }}
        with: &emulator-with
          api-level: 34
          arch: x86_64
          # AOSP's image, without Google Play services: on a fresh boot
          # they restart their process once a module is updated, at no set
          # time, and Android then kills every app holding a connection to
          # one of their providers. Android System WebView holds one to
          # their FontsProvider (its emoji font) for as long as it runs, so
          # the app under test was killed whenever the restart came during
          # the test (bats-lang/quire#220). The app uses nothing of Play
          # services
          target: default
          disable-animations: true
          # The screen: 320 dp wide, the narrowest width WCAG's Reflow
          # asks for (bats-lang/quire#265 was found at it), at a phone's
          # density of today: 960x1920 pixels at 480 dpi (3x), not the
          # AVD's default 320x640 at 160 dpi, whose screenshots are
          # blurred. The action has no input for the screen (`profile`
          # names only a hardware profile, and none is 320 dp at 3x), so
          # -skin sets the size and the AVD's config.ini, written after
          # the action makes it, the density
          # The action's own options, but the emulator's output goes to a
          # file, which is printed after the test (bats-lang/quire#258)
          # (the file is made first: the emulator does not create it)
          emulator-options: >-
            -no-window -gpu swiftshader_indirect -no-snapshot -noaudio -no-boot-anim
            -skin 960x1920
            ${{ inputs.emulator-extra-options }}
            -stdouterr-file ${{ runner.temp }}/emulator.log
          # The SDK pins again (android-sdk-pins, pwa#137): the action installed
          # the newest packages first, which replaced the pins if Google has
          # published newer ones
          pre-emulator-launch-script: |
            touch "$RUNNER_TEMP/emulator-installed"
            cd "$GITHUB_WORKSPACE/pwa/tools/android-sdk-pins" && "$GITHUB_WORKSPACE/sdk-pins-bin/android-sdk-pins" install /usr/local/lib/android/sdk
            touch "$RUNNER_TEMP/emulator.log"
            config="${ANDROID_AVD_HOME:-$HOME/.android/avd}/test.avd/config.ini" && sed -i '/^hw\.lcd\.\(width\|height\|density\)=/d' "$config" && printf 'hw.lcd.width=960\nhw.lcd.height=1920\nhw.lcd.density=480\n' >> "$config" && grep -E '^(hw\.lcd|skin)\.' "$config"
          # The screen is checked first, and named in a notice: a screen
          # other than 960x1920 at 480 dpi fails the job. A failed test
          # prints, while the emulator still runs, what the
          # app, Android and the screen were doing (smoke-evidence.sh, for
          # at most 10 minutes: an adb call that hangs does not hold the
          # job to its 6-hour limit)
          script: >-
            screen="$(adb shell wm size | tr -d '\r' | tr '\n' ' ')$(adb shell wm density | tr -d '\r' | tr '\n' ' ')";
            case "$screen" in *"Physical size: 960x1920"*"Physical density: 480"*) echo "::notice::The emulator's screen: $screen";;
            *) echo "::error::The emulator's screen is not 960x1920 at 480 dpi: $screen"; exit 1;; esac;
            if [ -n "$SMOKE_SHARE_FILE" ];
            then sh "app/${{ inputs.project-dir }}/smoke-test.sh" "$SMOKE_APK" "$SMOKE_TEXT" smoke "app/$SMOKE_SHARE_FILE" "$SMOKE_SHARE_TYPE" "$SMOKE_SHARE_TEXT";
            else sh "app/${{ inputs.project-dir }}/smoke-test.sh" "$SMOKE_APK" "$SMOKE_TEXT" smoke;
            fi
            || { timeout 600 sh "$RUNNER_TEMP/smoke-evidence.sh" "app/${{ inputs.project-dir }}/capacitor.config.json"; exit 1; }

      # The whole decision is the tool's (`android-sdk-pins decide`, which prints
      # again=true or again=false) and the second attempt runs on that output
      # alone; the step fails the job when the first attempt failed and is not
      # run again
      - name: Whether the emulator step runs again
        id: decide
        env:
          ANDROID_SMOKE_OUTCOME: ${{ steps.emulator1.outcome }}
          ANDROID_SMOKE_ATTEMPT: '1'
          ANDROID_SMOKE_ATTEMPTS: '2'
        run: |
          "$GITHUB_WORKSPACE/sdk-pins-bin/android-sdk-pins" decide "$RUNNER_TEMP/emulator-installed" >> "$GITHUB_OUTPUT"

      - name: Run the app on an emulator (attempt 2 of 2)
        id: emulator2
        if: ${{ steps.decide.outputs.again == 'true' }}
        uses: reactivecircus/android-emulator-runner@PIN
        env: *emulator-env
        with: *emulator-with

      # The emulator's own log: its graphics lines always (so a passing
      # run can be told from a failing one), all of it when the test failed
      - name: The emulator's log
        if: always()
        env:
          OUTCOME: ${{ steps.emulator2.outcome != 'skipped' && steps.emulator2.outcome || steps.emulator1.outcome }}
        run: |
          LOG="$RUNNER_TEMP/emulator.log"
          [ -f "$LOG" ] || { echo "No emulator log"; exit 0; }
          echo "== Graphics lines of the emulator's log"
          grep -nE 'ColorBuffer|GPU|gpu|[Gg]fxstream|[Vv]ulkan|GLES|EGL|OpenGL|SwiftShader|[Rr]ender' "$LOG" | tail -150 || true
          echo "ColorBuffer errors: $(grep -c 'ColorBuffer' "$LOG" || true)"
          if [ "$OUTCOME" != success ]; then
            echo "== The emulator's log (last 400 lines)"
            tail -400 "$LOG"
          fi
